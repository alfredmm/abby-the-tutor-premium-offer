  const observer = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) {
        e.target.classList.add('visible');
        // stagger children if they are for-cards or inc-cards
        const cards = e.target.querySelectorAll('.for-card, .inc-card, .promise-pill');
        cards.forEach((c, i) => {
          setTimeout(() => c.classList.add('visible'), i * 100);
        });
      }
    });
  }, { threshold: 0.1 });

  document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

  // Also observe individual cards
  const cardObserver = new IntersectionObserver((entries) => {
    entries.forEach((e, i) => {
      if (e.isIntersecting) {
        setTimeout(() => e.target.classList.add('visible'), 100);
      }
    });
  }, { threshold: 0.15 });
